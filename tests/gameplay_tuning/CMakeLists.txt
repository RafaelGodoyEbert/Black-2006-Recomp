cmake_minimum_required(VERSION 3.20)
project(black_gameplay_tuning C)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(gameplay-tuning-test test_gameplay.c)
target_include_directories(gameplay-tuning-test PRIVATE ../../third_party/xboxrecomp/src/video)
if(MSVC)
    target_compile_options(gameplay-tuning-test PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(gameplay-tuning-test PRIVATE -Wall -Wextra -Wpedantic -Werror -frounding-math)
    target_link_libraries(gameplay-tuning-test PRIVATE m)
endif()
enable_testing()
add_test(NAME gameplay-tuning-helper COMMAND gameplay-tuning-test)
add_test(NAME gameplay-tuning-generator COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_generator.py")
