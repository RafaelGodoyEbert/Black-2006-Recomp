cmake_minimum_required(VERSION 3.20)
project(black_aim_mode_test C)
if(NOT WIN32)
    message(FATAL_ERROR "The aim wrapper fixture requires Windows")
endif()
set(manual "${CMAKE_CURRENT_SOURCE_DIR}/../../src/recomp_manual.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manual}")
file(READ "${manual}" source)
string(REGEX MATCHALL "/[*] BEGIN BLACK_PC_AIM" starts "${source}")
string(REGEX MATCHALL "/[*] END BLACK_PC_AIM [*]/" ends "${source}")
list(LENGTH starts start_count)
list(LENGTH ends end_count)
if(NOT start_count EQUAL 1 OR NOT end_count EQUAL 1)
    message(FATAL_ERROR "Expected exactly one BLACK_PC_AIM block")
endif()
string(FIND "${source}" "/* BEGIN BLACK_PC_AIM" begin)
string(FIND "${source}" "/* END BLACK_PC_AIM */" end)
if(end LESS begin)
    message(FATAL_ERROR "Aim markers out of order")
endif()
math(EXPR length "${end} - ${begin} + 22")
string(SUBSTRING "${source}" ${begin} ${length} aim)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/aim.extracted.inc" "${aim}\n")
add_executable(aim-mode-test test_aim.c)
target_include_directories(aim-mode-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xboxrecomp/src/video")
target_compile_options(aim-mode-test PRIVATE /W4 /fp:strict /UNDEBUG)
enable_testing()
add_test(NAME aim-mode COMMAND aim-mode-test)
