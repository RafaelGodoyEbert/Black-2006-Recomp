cmake_minimum_required(VERSION 3.20)
project(black_lod_distance C)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xboxrecomp" CACHE PATH "Runtime source")
add_executable(lod-distance-test test_lod.c)
target_include_directories(lod-distance-test PRIVATE "${XBOXRECOMP_DIR}/src/video")
target_compile_definitions(lod-distance-test PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(lod-distance-test PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(lod-distance-test PRIVATE -Wall -Wextra -Wpedantic -Werror -frounding-math)
    target_link_libraries(lod-distance-test PRIVATE m)
endif()
enable_testing()
add_test(NAME lod-distance-helper COMMAND lod-distance-test)
add_test(NAME lod-distance-generator COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_generator.py")
